// STRIPE OLAP - COMPLETE STAR SCHEMA
// Dimensions  Facts and Analytical Aggregates

// ========================================
// DIMENSIONS (Base du Star Schema)
// ========================================

Table dim.customer {
	customer_key serial [pk, note: "surrogate key"]
	customer_id uuid [note: "business key, unique"]
	name varchar
	first_name varchar
	full_address text
	post_code varchar
	phone varchar
	email varchar
	created_at timestamptz
	updated_at timestamptz
}

Table dim.merchant {
	merchant_key serial [pk, note: "surrogate key"]
	merchant_id uuid [note : "business key, unique"]
	legal_name varchar
	full_address text
	post_code varchar
	phone varchar
	email varchar
	created_at timestamptz
	updated_at timestamptz
}

Table dim.product {
	product_key serial [pk, note : "surrogate key"]
	product_id uuid [note : "business key, unique"]
	name varchar
	description text
	price decimal
	category varchar
	is_active bool
	created_at timestamptz
	updated_at timestamptz
}

Table dim.payment_method {
	payment_method_key serial [pk, note : "surrogate key"]
	payment_method_id uuid [note : "business key, unique"]
	method_type varchar [note:  "card, bank_transfer, wallet"]
	card_brand varchar
	is_subscription bool
	is_active bool
	created_at timestamptz
}

Table dim.date {
	date_key int [pk, note: "YYYYMMDD format"]
	full_date date
	year int
	quarter int [note :"1-4"]
	month int [note: "1-12"]
	month_name varchar
	week int [note :"1-53"]
	day_of_month int
	day_of_week int [note : "1-7"]
	day_name varchar
	is_weekend bool
	is_holiday bool
}

Table dim.time {
	time_key int [pk, note: "HHMMSS format"]
	hour int [note : "0-23"]
	minute int [note : "0-59"]
	second int [note : "0-59"]
	time_of_day varchar [note : "Morning, Afternoon, Evening, Night"]
}

Table dim.location {
	location_key serial [pk, note:  "surrogate key"]
	location varchar [note : "original location string"]
	city varchar
	region varchar
	country varchar
	country_code varchar
}

// ========================================
// FACTS (Grain = 1 transaction)
// ========================================

Table fact.transactions {
	transaction_key serial [pk, not null]
	transaction_id uuid [note: 'fk']
	customer_key int [note: 'fk']
	merchant_key int [note: 'fk']
	product_key int [note: 'fk']
	payment_method_key int [note: 'fk']
	date_key int [note: 'fk']
	time_key int [note: 'fk']
	location_key int [note: 'fk']
	amount decimal 
	currency varchar
	status varchar 
	transaction_type varchar 
	device_type varchar
	is_successful bool
	is_refund bool
	created_at timestamptz
}

Table fact.fraud_scores {
	fraud_score_key serial pk
	fraud_score_id uuid [note: 'fk']
	transaction_key int [note: 'fk'] 
	customer_key int [note: 'fk']
	merchant_key int [note: 'fk']
	payment_method_key int [note: 'fk']
	date_key int [note: 'fk']
	time_key int [note: 'fk']
	fraud_score decimal
	risk_level varchar [note : "low, medium, high, critical"]
	model_version varchar
	manual_review_required bool
	scored_at timestamptz
}
